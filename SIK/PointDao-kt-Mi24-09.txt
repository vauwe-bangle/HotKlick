package de.softopus.drawpoint.data

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.Query
import kotlinx.coroutines.flow.Flow

@Dao
interface PointDao {
    @Query("SELECT * FROM points ORDER BY id ASC")
    fun getAllPoints(): Flow<List<PointEntity>>
    
    @Insert
    suspend fun insertPoint(point: PointEntity)
    
    @Query("DELETE FROM points")
    suspend fun deleteAllPoints()
    
    @Query("DELETE FROM points WHERE id = :pointId")
    suspend fun deletePointById(pointId: Int)
    
    @Query("SELECT COUNT(*) FROM points")
    suspend fun getPointCount(): Int
}
